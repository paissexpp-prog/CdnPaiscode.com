Dokumentasi API

Gunakan endpoint kami secara langsung ke dalam project Node.js Anda.

A. Upload File Lokal (Axios) 

const axios = require('axios');
const fs = require('fs');
const FormData = require('form-data');

async function uploadLocalFile(filePath) {
  try {
    const form = new FormData();
    // Masukkan file ke dalam field 'file'
    form.append('file', fs.createReadStream(filePath));

    const response = await axios.post(
      'https://cdn.paiscode.com/upload', 
      form, 
      { headers: { ...form.getHeaders() } }
    );

    console.log('✅ Berhasil:', response.data.data.url);
    return response.data.data.url;
  } catch (error) {
    console.error('❌ Gagal:', error.message);
  }
}

B. Upload via URL Gambar 

const axios = require('axios');

async function uploadFromUrl(imageUrl) {
  try {
    const response = await axios.post(
      'https://cdn.paiscode.com/upload', 
      { url: imageUrl }
    );

    console.log('✅ Berhasil CDN:', response.data.data.url);
    return response.data.data.url;
  } catch (error) {
    console.error('❌ Gagal:', error.message);
  }
}
